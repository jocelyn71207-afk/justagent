同步資料流（每次 triggerIntegrationSync）
// 1. 取 Database 所有 rows（含 Properties）
POST /v1/databases/{databaseId}/query
  → pages[]: { id, properties: { Name, Category, Tags, ... } }

// 2. 每個 page 取 blocks（Page Body）
GET /v1/blocks/{pageId}/children
  → blocks[]: { type: 'paragraph'|'heading_2'|'bulleted_list_item'|... }

// 3. Blocks → Markdown 轉換
heading_2 → ## title
paragraph → text
bulleted_list_item → - item

// 4. 建立/更新知識條目
for each page:
  title = properties[titleProp].title[0].plain_text
  content = blocksToMarkdown(blocks)
  category = properties[categoryProp]?.select?.name
  tags = properties[tagsProp]?.multi_select?.map(t => t.name)
  createOrUpdateKnowledgeDraft(title, content, category, tags)
