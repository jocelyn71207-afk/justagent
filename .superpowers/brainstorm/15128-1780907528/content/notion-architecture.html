通用整合框架架構

IntegrationSource 抽象層 · Notion 為第一個實作

資料模型

interface IntegrationSource {
  id: string
  type: 'NOTION' | 'GOOGLE_DRIVE' | 'SLACK'
  name: string
  enabled: boolean
  schedule: 'MANUAL' | 'DAILY' | 'WEEKLY'
  lastSyncAt: string | null
  lastSyncStatus: 'SUCCESS' | 'FAILED' | null
  lastSyncCount: number
  lastSyncError: string | null
  config: NotionConfig | GoogleDriveConfig | SlackConfig
}

// Notion 專用設定
interface NotionConfig {
  apiKey: string
  databaseId: string
  titleProp: string
  categoryProp?: string
  tagsProp?: string
  includePageBody: true
  contentPropFallback?: string
}

1. 呼叫 /databases/{id}/query 取 rows
